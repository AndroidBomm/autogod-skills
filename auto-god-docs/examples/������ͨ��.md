## $http 网络请求

```javascript
// GET 请求
$http.get("https://api.example.com/data", (resp) => {
    let data = resp.body.json();
    log("返回数据:", data);
}, (err) => {
    log("请求失败:", err);
});

// POST 请求
$http.post("https://api.example.com/submit", {
    name: "张三",
    age: 18
}, (resp) => {
    log("提交成功:", resp.body.string());
});

// POST JSON
$http.post("https://api.example.com/api", JSON.stringify(data), {
    contentType: "application/json",
    header: {
        "Authorization": "Bearer token123"
    }
}, (resp) => {
    log("响应:", resp.body.json());
});

// 文件下载（带进度）
$http.download("https://example.com/file.zip", "/sdcard/file.zip",
    (cur, total, percent) => {
        log("下载进度:", percent + "%");
    },
    () => {
        log("下载完成");
    },
    (err) => {
        log("下载失败:", err);
    }
);

// 上传文件
$http.upload("https://api.example.com/upload", "/sdcard/photo.png", {
    fileKey: "file",
    params: {
        type: "avatar"
    }
}, (resp) => {
    log("上传成功:", resp.body.string());
});
```

## $ws WebSocket

```javascript
// 创建连接
let ws = $ws.newWebSocket("wss://example.com/ws");

// 连接成功
ws.onOpen(() => {
    log("连接成功");
    ws.send("Hello Server");
});

// 收到消息
ws.onMessage((msg) => {
    log("收到:", msg);
});

// 连接关闭
ws.onClose(() => {
    log("连接关闭");
});

// 连接错误
ws.onError((err) => {
    log("错误:", err);
});

// 关闭连接
ws.close();
```
