## $alert 对话框

```javascript
// 基本使用
$alert.create()
.title("标题")
.msg("我是信息内容")
.but1("按钮1")
.but2("按钮2")
.but3("按钮3")
.show();

// 单选对话框
$alert.create()
.title("单选")
.items(["选项1","选项2","选项3"], (di, which) => {
    info("选中了:" + which);
})
.but3("取消")
.show();

// 单选按钮（带默认值）
$alert.create()
.title("单选")
.oneItems(["选项1","选项2","选项3"], 0, (di, which) => {
    info("选中了:" + which);
})
.but3("取消")
.show();

// 多选对话框
$alert.create()
.title("多选")
.moreItems(["选项1","选项2","选项3"], [0, 2], (di, which) => {
    info("选中了:" + which);
})
.but3("取消")
.show();
```

## $ui 界面

```javascript
// 创建布局
let ui = $ui.layout(`
<ui>
    <statusbar/>
    <appbar back="true" id="mAppbar" title="标题" w="max"/>
    <linear dir="v" w="max" h="max" padding="10">
        <button id="mBtn" text="点击" w="max"/>
    </linear>
</ui>
`);
ui.show();

// 返回按钮
ui.id("mAppbar").back(() => {
    ui.finish();
});

// 按钮点击
ui.id("mBtn").click(() => {
    toast("点击了按钮");
});
```
