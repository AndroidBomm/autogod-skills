# 代码示例 - 系统与设备

## $act 手势动作

```javascript
// 获取无障碍权限
let has = $act.hasPermit();
if (!has) {
    $act.getPermit();
}

// 点击
$act.click(400, 348);

// 长按
$act.press(400, 348);

// 滑动（下拉通知栏）
$act.move(400, 0, 400, 800, 1000);

// 双指操作
let g1 = [504, 583, 504, 583, 0, 1000];  // 先长按
let g2 = [504, 583, 100, 200, 0, 1000];  // 再滑动
$act.gesture([g1, g2]);

// 获取节点
let node = $act.selector()
    .text("按钮")
    .waitFirst(5000);

if (node != null) {
    node.click();
}
```

## $screen 屏幕

```javascript
// 获取截屏权限
$screen.getPermit();

// 截屏
let img = $screen.getScreen();

// 按区域截屏
let region = $screen.getScreen(0, 0, 500, 500);

// 获取屏幕尺寸
let w = $screen.getWidth();
let h = $screen.getHeight();

// 亮度控制
$screen.setBrightness(200);
let brightness = $screen.getBrightness();

// 强制横屏/竖屏
$screen.forceOrientation("portrait");
$screen.forceOrientation("landscape");
$screen.recoverOrientation();  // 恢复自动旋转
```

## $device 设备信息

```javascript
// 屏幕信息
let info = $device.info();
log("屏幕:", info.screen.width, "x", info.screen.height);

// 电池
let battery = $device.getBattery();
log("电量:", battery);

// 内存
let memory = $device.getMemory();
log("剩余内存:", memory);

// 音量
let vol = $device.getMusicVolume();
$device.setMusicVolume(10);

// 震动
$device.vibrate(1000);
```
