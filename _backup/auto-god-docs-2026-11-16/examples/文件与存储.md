## $file 文件操作

```javascript
// 读取文件
let content = $file.read("/sdcard/test.txt");

// 写入文件（内容在前，路径在后）
$file.write("Hello World", "/sdcard/test.txt");

// 追加内容
$file.add("追加的内容\n", "/sdcard/test.txt");

// 删除文件
$file.del("/sdcard/test.txt");

// 判断是否存在
let exists = $file.exist("/sdcard/test.txt");

// 文件信息
let size = $file.size("/sdcard/test.txt");
let name = $file.name("/sdcard/test.txt");
let ext = $file.ext("/sdcard/test.txt");

// 目录操作
$file.mkdir("/sdcard/myfolder");
$file.ensureDir("/sdcard/a/b/c");
let list = $file.ls("/sdcard");

// 递归遍历
$file.loopFiles("/sdcard", (file) => {
    log("找到文件:", file.getPath());
    return true;
});

// 拷贝
$file.copy("/sdcard/source.txt", "/sdcard/dest.txt");

// 移动
$file.move("/sdcard/old.txt", "/sdcard/new.txt");

// 获取存储目录
let sd = $file.sd();
log("SD卡路径:", sd);

// 路径拼接
let path = $file.join(sd, "folder", "file.txt");
```

## $storage 键值存储

```javascript
// 存入数据
$storage.put("name", "张三");
$storage.put("age", 18);
$storage.put("data", {key: "value"});

// 读取数据
let name = $storage.get("name");
let age = $storage.get("age", 0);  // 带默认值
let data = $storage.get("data");

// 删除
$storage.remove("name");

// 清空
$storage.clear();

// 是否包含
let has = $storage.contains("name");
```

## $sqlite 数据库

```javascript
// 打开/创建数据库
let db = $sqlite.open("/sdcard/data.db");

// 执行 SQL
db.execute("CREATE TABLE IF NOT EXISTS user (id INT, name TEXT)");

// 插入
db.execute("INSERT INTO user VALUES (?, ?)", [1, "张三"]);

// 查询
let cursor = db.rawQuery("SELECT * FROM user", []);
while (cursor.next()) {
    let id = cursor.getInt(0);
    let name = cursor.getString(1);
    log(id, name);
}
cursor.close();

// 关闭
db.close();
```
