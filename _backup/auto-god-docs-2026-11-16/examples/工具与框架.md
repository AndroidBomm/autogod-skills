## $thread 多线程

```javascript
// 创建线程
$thread.run(() => {
    // 在子线程执行
    let img = $screen.getScreen();
    let result = $ocr.detect(img);
    log("识别结果:", result);
});

// 定时循环
$thread.loop(() => {
    // 每3秒执行一次
    checkStatus();
}, 3000);

// 倒计时锁
let lock = new CdLock(3);
$thread.run(() => {
    // 任务1
    lock.countDown();
});
$thread.run(() => {
    // 任务2
    lock.countDown();
});
$thread.run(() => {
    // 任务3
    lock.countDown();
});
lock.wait();  // 等待所有任务完成
log("全部任务完成");
```

## $tip 对话框

```javascript
// 输入对话框
$tip.input("标题", "请输入名称", "默认值", (text) => {
    if (text) {
        log("输入:", text);
    }
});

// 单选对话框
$tip.one("请选择", ["选项1", "选项2", "选项3"], (index, text) => {
    log("选中:", text);
});

// 多选对话框
$tip.more("请选择", ["选项1", "选项2", "选项3"], [0, 2], (selected) => {
    log("选中:", selected);
});

// 日历选择
$tip.calendar((year, month, day) => {
    log("日期:", year, month, day);
});

// 时间选择
$tip.time((hour, minute) => {
    log("时间:", hour, minute);
});

// 确认对话框
$tip.show("提示", "确定要删除吗?", () => {
    log("用户确认");
});
```

## $yolo 目标检测

```javascript
// 加载模型
let model = $yolo.load({
    model: "/sdcard/model.onnx",
    names: "/sdcard/names.txt"
});

// 检测图片
let img = $screen.getScreen();
let results = model.detect(img);

// 遍历结果
for (let obj of results) {
    log("检测到:", obj.label, "置信度:", obj.score);
    $draw.rect(obj.x, obj.y, obj.w, obj.h);
}

// 释放模型
model.release();
```
