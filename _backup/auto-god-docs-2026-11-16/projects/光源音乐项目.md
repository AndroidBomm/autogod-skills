## 核心知识点

### 1. 列表绑定 ($ui + ls)

```javascript
let ls = ui.id("list");

// 绑定列表项布局
ls.bindHolder((holder, data, position) => {
    // data 是文件路径字符串
    let fileMainName = $file.mainName(data);
    holder.id("name").setText(fileMainName);
    
    // 点击事件
    holder.id("card").click(() => {
        let content = $file.read(data);
        let obj = JSON.parse(content);
        // 处理数据...
    });
    
    // 长按删除
    holder.id("card").longClick(() => {
        $tip.show("删除文件", "确认删除:\n" + data, () => {
            ls.del(position);  // 从列表移除
            $file.del(data);   // 删除文件
        });
    });
});

// 刷新列表
let flushList = function() {
    $thread.run(() => {
        $file.ensureDir("/sdcard/光遇音乐");
        let list = $file.ls("/sdcard/光遇音乐", (path) => {
            return path.endsWith(".json");
        });
        ls.flush(list);  // 更新列表数据
    });
}
```

### 4. 文件读写

```javascript
// 读 JSON 配置文件
function loadConfig(path) {
    if ($file.exist(path)) {
        let content = $file.read(path);
        return JSON.parse(content);
    }
    return {};
}

// 写 JSON 配置文件
function saveConfig(path, data) {
    $file.write(JSON.stringify(data), path);
}
```
