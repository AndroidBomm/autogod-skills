### 4. $floaty 系统悬浮窗

```javascript
// 创建系统级悬浮窗
let floaty = $floaty.newSys(`
<ui>
    <frame id="root" w="300dp" h="400dp" bg="#F0F0F0">
        <linear dir="v" w="max" h="max">
            <text id="title" text="悬浮窗" textSize="16" padding="10" />
            <button id="btnAction" text="执行操作" margin="10" />
            <button id="btnClose" text="关闭" margin="10" />
        </linear>
    </frame>
</ui>
`);

// ⚠️ SysFloaty 没有 getX()/getY()，必须用 JS 变量跟踪位置
let winX = 100, winY = 200;
floaty.setXY(winX, winY);

// 获取控件并绑定事件
floaty.id("btnClose").click(() => {
    floaty.close();
});
```

### 5. 触摸拖动实现

```javascript
// 触摸拖动悬浮窗
let startX = 0, startY = 0;

rootTouch.click((e) => {
    // 处理点击事件
});

rootTouch.longClick((e) => {
    // 处理长按
});

rootTouch.startDrag((e) => {
    startX = e.x;
    startY = e.y;
    return true;  // 开始拖动
});

rootTouch.drag((e) => {
    let dx = e.x - startX;
    let dy = e.y - startY;
    // 用跟踪变量，SysFloaty 没有 getX()/getY()
    winX += dx;
    winY += dy;
    floaty.setXY(winX, winY);
    startX = e.x;
    startY = e.y;
});
```

### 6. 响应式布局（横竖屏）

```javascript
let $layout_ui = function(ui) {
    function 获取屏幕方向() {
        let w = $screen.getWidth();
        let h = $screen.getHeight();
        return w > h ? "横屏" : "竖屏";
    }
    
    function 设置UI位置(方向) {
        let w = $screen.getWidth();
        let h = $screen.getHeight();
        if (方向 === "横屏") {
            ui.setXY(w * 0.1, h * 0.01);
            ui.setWH(w * 0.8, h * 0.9);
        } else {
            ui.setXY(w * 0.05, h * 0.01);
            ui.setWH(w * 0.9, h * 0.8);
        }
        ui.touch(true);
    }
    
    设置UI位置(获取屏幕方向());
    
    // 监听屏幕旋转
    floaty.onScreenChange((screenInfo) => {
        $ui.run(() => { 设置UI位置(获取屏幕方向()); });
    });
}
```
