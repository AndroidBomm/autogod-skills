##### 简要描述

-EC已经有二维码识别了,所以做了个条形码识别

##### 用到的方法
- ` binaryzationMat `

- ` close `

- ` open `

- ` getContoursRect `

- ` cropMat `

##### 代码示例
```javascript

    //用于识别条形码的对象
    let zxing= new  com.plugin.WIVGsmvllC.PluginClz(context); //根据自己的路径进行修改

    let srcMat = imageHandler.readResMat("条形码7.png");

    let 反向二值化=imageHandler.binaryzationMat(srcMat,69,255,1);

    let 闭操作=imageHandler.close(反向二值化,0,15,3);

    let 开操作=imageHandler.open(闭操作,0,5,30);

    let rects=imageHandler.getContoursRect(开操作,0,2);

    //画矩形 演示用

     let resultMat=imageHandler.drawRect2(srcMat,rects);
    logd(imageHandler.saveMat(resultMat,"/storage/emulated/0/Pictures/mat.png"));
    saveImg(srcMat,"画矩形");

    //进行识别
    for (let i = rects.length-1; i >=0; i--) {
        let 条形码=imageHandler.cropMat(srcMat,rects[i].left-5,rects[i].top,rects[i].right-5,rects[i].bottom)
        let result= zxing.decode2(条形码);
        imageHandler.recycle(条形码); //释放截图
        logd("识别结果为:"+result);
    }

    //图片释放代码略
    imageHandler.recycle(srcMat,resultMat,反向二值化,闭操作,开操作);

```
