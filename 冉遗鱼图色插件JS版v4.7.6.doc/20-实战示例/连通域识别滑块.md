##### 简要描述

-通过对连通域的属性进行滑块识别

##### 用到的方法
- 剪切` cropMat `

- 高斯模糊` gaussianBlur `

- 锐化` filter2D `

- 二值化` binaryzationMat `

- 开操作` open `

- 连通域` connectedComponentsWithStats `


##### 代码示例
```javascript

    	//一共19张测试图片 循环19次  只适合这种大小的图片(我也不知道什么分辨率下的,别人给的图片,图片会发群里)
  	  for(let j=1;j<=19;j++){
        let mat=imageHandler.readResMat(j+".png");
        if(!mat){
            logd("第"+j+"张图片为空");
            continue;
        }

        //只识别后半部分,所以去掉头部
        let 后半部分=imageHandler.cropMat(mat,mat.getWidth()/5,0,mat.getWidth(),mat.getHeight())


        //高斯模糊 是为去点有些滑块内部的白点,测试后核大小5合适,不然不利于之后的反向二值化的开操作
        //之前开操作的核大小是21,现在开操作的核大小改成5了, 不用高斯模糊也没事
        let 高斯模糊=imageHandler.gaussianBlur(后半部分,5);

        //提高对比度(亮的地方越亮)
        let 锐化=imageHandler.filter2D(高斯模糊);

        //使目标变成白色 为后面的 连通域使用(连通域只计算白色)
        let 反向二值化=imageHandler.binaryzationMat(锐化,150,255,1);

        //去 噪点
        let 开操作=imageHandler.open(反向二值化,0,5,5);

        //使用连通域 获取所有白色块的信息
        let 连通域=imageHandler.connectedComponentsWithStats(开操作,4);

        if(连通域){
            for (let i = 0; i < 连通域.length; i++) {

                logd(连通域[i].whiteArea);
                //首先 面积要符合滑块的大小
                if (连通域[i].whiteArea>7000 && 连通域[i].whiteArea<11000) {
                    let w=连通域[i].rect.bottom-连通域[i].rect.top;
                    let h=连通域[i].rect.right-连通域[i].rect.left;

                    //其次宽高 以及 宽高比要符合要求
                    if(w*1.0/h<1.3 && w*1.0/h>0.7 && w<150 && h<150){

                        //在原图上画出最终结果
                        let r3=imageHandler.drawRect(mat,连通域[i].rect.left+mat.getWidth()/5,连通域[i].rect.top,连通域[i].rect.right+mat.getWidth()/5,连通域[i].rect.bottom);
                        //保存
                        imageHandler.saveMat(r3,"/storage/emulated/0/Pictures/第"+j+"张滑块结果 第"+i+".png")
                        imageHandler.recycle(r3);

                        break; //我们这里只需要找到一个缺口
                    }
                }
            }
        }

        imageHandler.recycle(mat,后半部分,高斯模糊,锐化,反向二值化,开操作)


    }



```
