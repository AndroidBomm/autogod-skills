##### 用到的方法
- ` binaryzationMat `

- ` getContoursRect `

- ` cropMat `

- ` cmpBit `


##### 自作字库代码示例
```javascript


    let mat=imageHandler.readResMat("字库1.png");
    //让白底黑字变成黑底白字
    let 反向二值化=imageHandler.binaryzationMat(mat,127,255,1);
    //框选出数字
    let rects=imageHandler.getContoursRect(反向二值化,0,2,null);

    //按x轴进行从小到大排序
    rects.sort( (a,b)=>{return a.left>b.left;})

    //把字切割下来当字库
    for (let i = 0; i < rects.length; i++) {
        let temp=imageHandler.cropMat(反向二值化,rects[i].left,rects[i].top,rects[i].right,rects[i].bottom);
        imageHandler.saveMat(temp,"/storage/emulated/0/Pictures/数字"+i+".png");
        imageHandler.recycle(temp);
    }

    imageHandler.recycle(mat,反向二值化);



```

##### 数字识别代码示例
```javascript

    //先把制作好的字库图片加载进来
    let fontImg=[];
    for (let i = 0; i < 10; i++) {
        fontImg[i]=imageHandler.readResMat("数字"+i+".png");
    }
    let mat=imageHandler.readResMat("字库测试.png");
    //让黑字变白字
    let 反向二值化=imageHandler.binaryzationMat(mat,127,255,1);
    //框选出数字
    let rects=imageHandler.getContoursRect(反向二值化,0,2,null);
    //按x轴进行从小到大排序
    rects.sort( (a,b)=>{return a.left>b.left;})

    let 结果字符串="";
    //把字切割下来一样对比
    for (let i = 0; i < rects.length; i++) {
        let temp=imageHandler.cropMat(反向二值化,rects[i].left,rects[i].top,rects[i].right,rects[i].bottom);
        let 相似度=0;
        let 数字;
        for (let j = 0; j <fontImg.length; j++) {
            let r=imageHandler.cmpBit(fontImg[j],temp);
            if(r>相似度){ //如果相似度更高
                相似度=r;
                数字=j;
            }
        }
        if(相似度>0.8){ //调整到适当的阈值
            结果字符串+=数字;
        }
        imageHandler.recycle(temp);
    }
    logd("识别结果:"+结果字符串);

    imageHandler.recycle(fontImg,mat,反向二值化);
```
