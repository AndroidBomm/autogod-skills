##### 简要描述

-使用感知哈希值识别背包物品，也不排除哈希值一样的可能 比如两个目标只是局部不一样

##### 用到的方法
- ` cropMat `

- ` binaryzationMat `

- ` getContoursRect `

- ` drawRect `

- ` pHashCode `

- ` hanMingDistance `

##### 代码示例
```javascript

function  切割目标(mat,rect){

    let x_dis=69;//目标X轴间隔为69
    let y_dis=69;//目标Y轴间隔为69

    let w=rect.right-rect.left;//目标宽度
    let h=rect.bottom-rect.top;//目标高度
    let x=rect.left;
    let y=rect.top;

    //将开始坐标移至左上角 方便后续切割
    while (x-x_dis>=0){
        x-=x_dis;
    }
    while (y-y_dis>=0){
        y-=y_dis;
    }

    function  obj(mat, left, top, right, bottom){
             this.srcMat=mat;
            this.left=left;
            this.top=top;
            this.right=right;
            this.bottom=bottom;
    };

    let targets=[];//目标集合

    let sx=x;
    let sy=y;
    //按从左到右 从上到下顺序切割目标
    while (sy+h<mat.getHeight()){
        while (sx+w<mat.getWidth()){
            let 目标=imageHandler.cropMat(mat,sx,sy,sx+w,sy+h);
            targets.push( new obj(目标,sx,sy,sx+w,sy+h) );
            sx+=x_dis;
        }
        sx=x;//坐标移至最左侧
        sy+=y_dis;//下一行
    }
    return targets
}

//涂鸦 去掉每个目标的 左上角和 右下角数字
function  涂鸦目标(targets){
    for (let i = 0; i < targets.length; i++) {
        let mat=imageHandler.drawRect(targets[i].srcMat,0,0,24,18,"#000000",-1)
        mat=imageHandler.drawRect(mat,34,45,mat.getWidth(),mat.getHeight(),"#000000",-1)
        targets[i].srcMat=mat;
    }
    return targets;
}


function 哈希表的制作(){

    //截图或者读取图片
    let 游戏截图 = imageHandler.readResMat("beibao.png");
    let 背包区域=  imageHandler.cropMat(游戏截图,924,117,1208,506);
    let 二值化= imageHandler.binaryzationMat(背包区域,32);
    let rects=imageHandler.getContoursRect(二值化,0,2);

    let rect;
    if(rects){
        for (let i = 0; i < rects.length; i++) {
            //通过观察 目标宽高为64
            if(rects[i].bottom-rects[i].top==64 && rects[i].right-rects[i].left==64 ){
                rect=rects[i] //找一个 目标做锚点切割其他目标
                break;
            }
        }
    }
    let targets=切割目标(背包区域,rect);
    targets=涂鸦目标(targets);

    for (let i = 0; i < targets.length; i++) {
        let hashValue=imageHandler.pHashCode(targets[i].srcMat);
        //用 数字+|号为了方便按照图片写名称
        file.appendLine(i+"|"+hashValue, "/storage/emulated/0/Pictures/test.txt");
        imageHandler.saveMat(targets[i].srcMat,"/storage/emulated/0/Pictures/"+i+".png","png",100);
    }
    //回收所有图片
    imageHandler.recycleAllMat();

}


function 识别物品(){

    let names=[]; //物品名称
    let hashCode=[];//哈希表值

    let arr = file.readAllLines("/storage/emulated/0/Pictures/test.txt");
    for(let str of arr){
        let temp=str.split("|");
        names.push(temp[0]);//名称
        hashCode.push(temp[1]);//哈希值
    }


    //截图或者读取图片
    let 游戏截图 = imageHandler.readResMat("beibao.png");
    let 背包区域=  imageHandler.cropMat(游戏截图,924,117,1208,506);
    let 二值化= imageHandler.binaryzationMat(背包区域,32);
    let rects=imageHandler.getContoursRect(二值化,0,2);

    let rect;
    if(rects){
        for (let i = 0; i < rects.length; i++) {
            //通过观察 目标宽高为64
            if(rects[i].bottom-rects[i].top==64 && rects[i].right-rects[i].left==64 ){
                rect=rects[i] //找一个 目标做锚点切割其他目标
                //break;
            }
        }
    }


    let targets=切割目标(背包区域,rect);//targets 里的坐标只是相对于 背包区域 的
    targets=涂鸦目标(targets);


    for (let obj of  targets){
        let hashValue=imageHandler.pHashCode(obj.srcMat); //生成哈希值
        //哈希值比较
        let maxValue=0;
        let index=0;
        for (let i = 0; i < hashCode.length; i++) {
            let value = imageHandler.hanMingDistance(hashValue,hashCode[i]);//汉明距离

            if( (1-value/64) >maxValue){ // 也可以把value 归一化到 0~1   (1-value/64)   其中64是字符串长度    1：完全一样 0：完全不一样
                maxValue=(1-value/64);
                index=i;
            }
        }
        logd("目标为"+names[index]+" 相似度："+maxValue+" 在全屏位置的坐标 x:"+(obj.left+924)+",y:"+ (obj.top+117));//924 117 是背包区域图片相对于全屏的起点
    }
}





function main() {

    //哈希表的制作();
    识别物品();
}
```
