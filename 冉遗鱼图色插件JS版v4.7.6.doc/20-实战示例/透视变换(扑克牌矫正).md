##### 简要描述

-自动获取扑克牌四角的坐标并矫正扑克牌

##### 用到的方法

- ` binaryzationMat `

- ` polygonDetection `

- ` getApproxPolyDPPoint `

- ` warpPerspective `

##### 代码示例
```javascript


    let cardWidth = 250;//转换后的宽带
    let cardHeight = 350;//转换后的高度
    //转换后的坐标
    let dstPoints=[[0,0],[0,cardHeight],[cardWidth,cardHeight],[cardWidth,0]];

    //读取势力图片
    let src=imageHandler.readResMat("cards.jpg");
    //二值化
    let bin=imageHandler.binaryzationMat(src,180);
    //获取所有四边形的索引
    let indexes=imageHandler.polygonDetection(bin,4,0,200,-1,0,2,-1);

    if(indexes){
        //获取索引所在的四边形的角点坐标
        let points=imageHandler.getApproxPolyDPPoint(bin,0,2,indexes);

        for (let i = 0; i <points.length ; i++) {
            logd(JSON.stringify(points[i]))
            //根据扑克牌四边的中间点的距离 来判断扑克牌的方向 并进行坐标顺序改正
            {
                let p1={x:(points[i][0].x+points[i][1].x)/2,y:(points[i][0].y+points[i][1].y)/2 };//第一边的中间点坐标
                let p2={x:(points[i][1].x+points[i][2].x)/2,y:(points[i][1].y+points[i][2].y)/2 };//第二边的中间点坐标
                let p3={x:(points[i][2].x+points[i][3].x)/2,y:(points[i][2].y+points[i][3].y)/2 };//第三边的中间点坐标
                let p4={x:(points[i][3].x+points[i][0].x)/2,y:(points[i][3].y+points[i][0].y)/2 };//第四边的中间点坐标

                let  distance1= Math.pow(p1.x-p3.x,2)+Math.pow(p1.y-p3.y,2);//第一边和第三边的距离（平方）
                let  distance2= Math.pow(p2.x-p4.x,2)+Math.pow(p2.y-p4.y,2);//第二边和第四边的距离（平方）
                if(distance1>distance2){
                    points[i]=[points[i][1],points[i][2],points[i][3],points[i][0]]; //如果方向不对，改一下points[i]点的顺序
                }
            }
            //透视变换
            let result=imageHandler.warpPerspective(src,points[i],dstPoints);
            imageHandler.saveMat(result, "/storage/emulated/0/Pictures/"+i+"结果.png");
            imageHandler.recycle(result);
        }
    }
    imageHandler.recycle(src,bin);



```
