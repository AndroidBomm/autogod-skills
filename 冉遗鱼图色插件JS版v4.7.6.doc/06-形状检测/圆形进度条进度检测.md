##### 简要描述

- 检测圆形进度条的进度,0~359 (刘百浲大佬提供)

##### 方法名
- ` getCircleBarValue `
##### 参数

|参数名|必选|类型|说明|
|:----    |:---|:----- |-----   |
|mat |是  |Mat |原图   |
|color |是  |string |颜色字符串 例如:"#FFFFFF-0x101010"   |
|x |是  |int |圆心坐标 x   |
|y |是  |int |圆心坐标 y   |
|r |是  |int |圆形半径 r   |
|startAngle |否  |int |开始角度0~359,以12点钟为0度起点, -1表示自动计算起点,默认-1;  |
|endAngle |否  |int |结束角度1~360,默认360; |

##### 返回值说明 

|类型|说明|
|:-----  |:-----|
|int   |进度,0~359;  |

##### 代码示例
```javascript
    //例子1
    let 进度圆节点=id("com.ss.android.ugc.aweme.lite:id/bfw").getOneNodeInfo(0);
    if(进度圆节点){
        //获取圆心
        let centerP=进度圆节点.bounds.center();
        //获取半径 半径要比实际的短一些才能真正取到颜色
        let r=(进度圆节点.bounds.right-进度圆节点.bounds.left)/2 -2;
        let srcMat1=imageHandler.captureScreenMat();
        let 进度=imageHandler.getCircleBarValue(srcMat1,"#FFDB73-0x151515",centerP.x,centerP.y,r,0,360);
        logd("当前进度为:"+进度);
        imageHandler.recycle(srcMat1);
    }


    //例子2
    let srcMat=imageHandler.readResMat("jindu.png");
    let 进度=imageHandler.getCircleBarValue(srcMat,"#CDC8B0-0x151515", 165,280,38);
    logd("例子2进度:"+进度);
    imageHandler.recycle(srcMat)

```
