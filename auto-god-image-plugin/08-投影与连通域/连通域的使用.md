##### 简要描述

-识别出文字位置

##### 用到的方法
- ` 二值化 binaryzationMat `

- ` 腐蚀 dilate `

- ` 膨胀 erode `

- ` 连通域状态 connectedComponentsWithStats `

##### 代码示例
```javascript

  //读取了一张  图片
    let 原图=imageHandler.readResMat("连通域2.png");
    //反向二值化,成为白字黑底
    let 二值化=imageHandler.binaryzationMat(原图,127,255,1);
    //闭操作
    let 膨胀=imageHandler.dilate(二值化,0,12,3);
    let 腐蚀=imageHandler.erode(膨胀,0,10,3);

    let stats=imageHandler.connectedComponentsWithStats(腐蚀,4);
    if(stats){
        let temp=imageHandler.copyTo(原图,null); //拷贝一份原图 用来做画
        for (let i = 0; i < stats.length; i++) {
            if(stats[i].rect.bottom-stats[i].rect.top>20){ //满足一定高度才是字 而不是分界线

                let temp2 =imageHandler.drawRect2(temp,stats[i].rect);
                //回收就图片 赋予新图片
                imageHandler.recycle(temp);
                temp=temp2;
            }
        }
        //这是个自定义函数,不在插件里
        imageHandler.saveMat(temp,"/storage/emulated/0/Pictures/结果.png");
        imageHandler.recycle(temp);
    }

    imageHandler.recycle(原图,二值化,膨胀,腐蚀);


```
