##### 简要描述

-一个K歌滑块

##### 用到的方法
- ` cropMat `

- ` getHSV `

- ` inRange `

- ` doubleBinaryzation `

- ` findImg `

##### 代码示例
```javascript

    let img=imageHandler.readResMat("tu1.png");
    //对图片进行分割
    let 前半部分=imageHandler.cropMat(img,0,0,img.getWidth()/5,img.getHeight());
    let 后半部分=imageHandler.cropMat(img,img.getWidth()/5,0,img.getWidth(),img.getHeight());
    //对前半部分处理
    let 前HSV=imageHandler.getHSV(前半部分,-1);
    let 颜色提取=imageHandler.inRange(前HSV,[75,70,170],[90,170,255]);
    //对后半部分处理
    let 后H=imageHandler.getHSV(后半部分,2);
    let 双二值化=imageHandler.doubleBinaryzation(后H,50,140,255);

    let rect=imageHandler.findImg(双二值化,颜色提取,1,0.01,-1,0,0,0,1,颜色提取);

    logd(JSON.stringify(rect));

    logd("需要滑动的距离x:"+(rect[0].left+前半部分.getWidth()));
    //画出矩形框
    let  resultImg=imageHandler.drawRect(img,rect[0].left+img.getWidth()/5,rect[0].top,rect[0].right+img.getWidth()/5,rect[0].bottom);
    imageHandler.saveMat(resultImg,"/storage/emulated/0/Pictures/resultImg.png");

    imageHandler.recycle(img,前半部分,后半部分,前HSV,颜色提取,后H,双二值化,resultImg);


```
