##### 简要描述

-找出所有合成目标的位置

##### 用到的方法
- ` cropMat `

- ` getHSV `

- ` inRange `

- ` open `

- ` close `

- ` binaryzationMat `

- ` getContoursRect `

- ` drawRect `

##### 代码示例
```javascript

    let mat=imageHandler.readResMat("youxi233.png");
    imageHandler.saveMat(mat,"原图");//保存图片

    let 裁剪=imageHandler.cropMat(mat,0,1015,1080,1661);
     imageHandler.saveMat(裁剪,"/storage/emulated/0/Pictures/裁剪.png");
    let 转HSV=imageHandler.getHSV(裁剪,-1);
     imageHandler.saveMat(转HSV,"/storage/emulated/0/Pictures/转HSV.png");
    let 颜色范围提取=imageHandler.inRange(转HSV,[75,138,183],[93,181,216]);
     imageHandler.saveMat(颜色范围提取,"/storage/emulated/0/Pictures/颜色范围提取.png");
    let 开操作=imageHandler.open(颜色范围提取,0,7,7);
     imageHandler.saveMat(开操作,"/storage/emulated/0/Pictures/开操作.png");
    let 闭操作=imageHandler.close(开操作,0,100,5);
     imageHandler.saveMat(闭操作,"/storage/emulated/0/Pictures/闭操作.png");
    let 闭操作2=imageHandler.close(闭操作,0,5,100);
     imageHandler.saveMat(闭操作2,"/storage/emulated/0/Pictures/闭操作2.png");
    let 闭操作3=imageHandler.close(闭操作2,0,100,5);
     imageHandler.saveMat(闭操作3,"/storage/emulated/0/Pictures/闭操作3.png");
    let 反向二值化=imageHandler.binaryzationMat(闭操作3,127,255,1);
     imageHandler.saveMat(反向二值化,"/storage/emulated/0/Pictures/反向二值化.png");

    let rects=imageHandler.getContoursRect(反向二值化,0,2);

    let 最终结果=imageHandler.drawRect2(裁剪,rects);

    imageHandler.saveMat(最终结果,"/storage/emulated/0/Pictures/最终结果.png");

    imageHandler.recycle(mat,裁剪,转HSV,颜色范围提取,开操作,闭操作,闭操作2,闭操作3,反向二值化,最终结果);


```
