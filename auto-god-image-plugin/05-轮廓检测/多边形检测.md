##### 简要描述

- 多边形检测函数(三角形,矩形,五边形,多边形,圆等凸包形状)

##### 方法名
- ` polygonDetection `
##### 参数

|参数名|必选|类型|说明|
|:----    |:---|:----- |-----   |
|mat |是  |Mat |二值化图像或者轮廓   |
|Angles |是  |int |目标角数/边数 ,如三角形是3 五边形是5 |
|compareMode |否 |int |比较方式,-1表示小于Angles,0表示等于Angles,1表示大于Angles,默认为0 |
|minArea |否 |int |目标的最小面积 默认为100|
|maxArea |否 |int |目标的最大面积 默认为-1,表示无上限 |
|mode |否 |int |定义轮廓的检索模式(可选值0,1,2) 默认为0,<br/>0:只检测最外围轮廓,包含在外围轮廓内的内围轮廓被忽略; <br/>1:检测所有的轮廓,包括内围、外围轮廓,但是检测到的轮廓不建立等级关系,hierarchy向量内所有元素的第3、第4个分量都会被置为-1;<br/>2:检测所有的轮廓,但所有轮廓只建立两个等级关系,外围为顶层,若外围内的内围轮廓还包含了其他的轮廓信息,则内围内的所有轮廓均归属于顶层 |
|method |否 |int |定义轮廓的近似方法(可选值1,2) 默认为2; <br/>1:保存物体边界上所有连续的轮廓点到contours向量内;<br/>2:仅保存轮廓的拐点信息,把所有轮廓拐点处的点保存入contours向量内,拐点与拐点之间直线段上的信息点不予保留;|
|epsilon |否 |double |轮廓近似逼近阈值（多边形拟合） 默认为 -1 表示目标周长的百分之一为阈值|
##### 返回值说明

|类型|说明|
|:-----  |:-----|
|int[]或者null   |返回的是引索数组,配合其他轮廓函数使用,比如轮廓重心,轮廓外接矩形等|

##### 代码示例
```javascript
    
  // 截图或读取图片
    let srcMat=imageHandler.readResMat("duobian.png");
    //二值化或者轮廓检测 此处以二值化为例
    let 二值化=imageHandler.binaryzationMat(srcMat,50,255,0);



    //多边形检测
    let 三角形引索=imageHandler.polygonDetection(二值化,3,0,100,-1,0,2,-1);
    if(三角形引索){
        //获取轮廓
        let 三角轮廓=imageHandler.getContours(二值化,0,2,三角形引索);
        logd(imageHandler.saveMat(三角轮廓,"/storage/emulated/0/Pictures/三角轮廓.png"));
        imageHandler.recycle(三角轮廓);
    }


    //获取大于三条边的图形
    //多边形检测
    let 多于三边的图形的引索=imageHandler.polygonDetection(二值化,3,1,100,-1,0,2,-1);
    if(多于三边的图形的引索){
        //获取轮廓
        let 多边形轮廓=imageHandler.getContours(二值化,0,2,多于三边的图形的引索);
        logd(imageHandler.saveMat(多边形轮廓,"/storage/emulated/0/Pictures/多边形轮廓.png"));
        imageHandler.recycle(多边形轮廓);
    }


    //获取圆
    //检测大于10边的图形
    let 检测圆形引索=imageHandler.polygonDetection(二值化,10,1,100,-1,0,2,-1);
    if(检测圆形引索){
        //获取轮廓
        let 圆形轮廓=imageHandler.getContours(二值化,0,2,检测圆形引索);
        logd(imageHandler.saveMat(圆形轮廓,"/storage/emulated/0/Pictures/圆形轮廓.png"));
        imageHandler.recycle(圆形轮廓);
    }

    imageHandler.recycle(srcMat,二值化);


```
