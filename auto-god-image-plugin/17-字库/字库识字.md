##### 简要描述

- 根据字库对未知文字进行识别
- 要求：只能识别单行文字(横向的)，二值化后无背景干扰。可与连通域配合使用，见实例里 连通域对文字的定位。

##### 方法名
- ` getWords `
##### 参数

|参数名|必选|类型|说明|
|:----    |:---|:----- |-----   |
|mat |是  |Mat |原图   |
|colorInfo |是  |string |颜色字符串  例如 <br> "#903A00-0x101010丨#000000-0x101010",如果不写就自动使用字库的颜色|
|sx |否  |int |起点x 默认0  |
|sy |否  |int |起点y 默认0  |
|ex |否  |int |终点x 默认0 |
|ey |否  |int |终点y 默认0 |
|threshold |否  |double |阈值 默认 0.9   |
|colorFilter |否  |boolean |是否用设定的colorInfo对字库进行筛选,默认 false;  如果是true,有助于提供效率(前提是你的字库的颜色于设定colorInfo一致)   |
|fontName |否  |string |指定字库名称 例如 myFontName，不写则使用默认字库   |

##### 返回值说明 

|类型|说明|
|:-----  |:-----|
|RectEx[]或null   |返回区域数组或null |

##### 代码示例
#####方式一 简单的使用方式

```javascript


    //截图或读取图片
    let srcMat=imageHandler.readResMat("ziku2.png");
    if(!srcMat){logd("没有这个图"); return; }
    //只需要设置一次
    let r=imageHandler.setDict("test.txt");
    logd("设置字库结果:"+r);
    let  rects=imageHandler.getWords(srcMat,"#903A00-0x101010|#000000-0x101010",-1,0,0,0,0.9);

    logd(JSON.stringify(rects));

    imageHandler.recycle(srcMat);

```



#####方式二 配合连通域使用方式
```javascript


    //截图或读取图片
    let srcMat=imageHandler.readResMat("2.png");
    if(!srcMat){logd("没有这个图"); return; }
    //只需要设置一次
    let r=imageHandler.setDict("ziku2.txt");
    logd("设置字库结果:"+r);

    //取出想要的颜色的二值化图
    let 颜色范围提取=imageHandler.inRange(srcMat,[255,0,10],[255,179,183]);

    let 闭操作=imageHandler.close(颜色范围提取,0,41,3);

    let 连通域=imageHandler.connectedComponentsWithStats(闭操作,4);

    if(连通域){
        for (let i = 0; i < 连通域.length; i++) {//此处还可以根据 面积,长度等过滤 提高效率

            let  rects=imageHandler.getWords(srcMat,"FFC5CA-0x101010|#F74C5A-0x084C48",连通域[i].rect.left,连通域[i].rect.top,连通域[i].rect.right,连通域[i].rect.bottom,0.9);

            if(rects){
                logd(JSON.stringify(rects));
            }
        }
    }

    imageHandler.recycle(srcMat,颜色范围提取,闭操作)


```
