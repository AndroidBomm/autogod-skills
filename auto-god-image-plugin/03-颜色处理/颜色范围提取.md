##### 简要描述

- 对指定颜色进行提取,可与HLS HSV图片一起使用.参数使用十进制颜色数组.<br>例#AA50BB-0x101010 转十进制为[170-16,80-16,187-16]~[170+16,80+16,187+16]<br>也可以直接看图色工具放大镜中的RGB数值

##### 方法名
- ` inRange `
##### 参数

|参数名|必选|类型|说明|
|:----    |:---|:----- |-----   |
|mat |是  |Mat |原图   |
|lowerColor |是  |int[] |颜色下限数组 例如[30,127,90];    |
|upperColor |是  |int[] |颜色上限数组 例如[90,200,130];   |

##### 返回值说明

|类型|说明|
|:-----  |:-----|
|Mat   |掩膜图片 即二值化图片   |

##### 代码示例
```javascript

    //提取红色部分的颜色
    let 红色部分=imageHandler.inRange(HSV,[109,149,80],[130,255,220]);
	
```

##### 代码运用

```javascript
    //截图或者读取图片
    let 指南针=imageHandler.readResMat("znz.png");
    
    //有用颜色不单一(如 红~深红),转成HSV方便处理
    let HSV=imageHandler.getHSV(指南针,-1);
    logd(imageHandler.saveMat(HSV,"/storage/emulated/0/Pictures/HSV.png"));


    //提取红色部分的颜色
    let 红色部分=imageHandler.inRange(HSV,[109,149,80],[130,255,220]);
    logd(imageHandler.saveMat(红色部分,"/storage/emulated/0/Pictures/红色部分.png"));

    //提取红色部分的颜色
    let 蓝色部分=imageHandler.inRange(HSV,[4,149,80],[30,255,220]);
    logd(imageHandler.saveMat(蓝色部分,"/storage/emulated/0/Pictures/蓝色部分.png"));

    //合并红色部分与蓝色部分的二值化图
    let 合并=imageHandler.bitwise_or(红色部分,蓝色部分);
    logd(imageHandler.saveMat(合并,"/storage/emulated/0/Pictures/合并.png"));
    
    //进行掩膜复制 上色
    let 掩膜复制=imageHandler.copyTo(指南针,合并);
    logd(imageHandler.saveMat(掩膜复制,"/storage/emulated/0/Pictures/掩膜复制.png"));

    imageHandler.recycle(指南针,HSV,红色部分,蓝色部分,合并,掩膜复制);
```
